local function GetDistanceRoot(Objective)
	if typeof(Objective) == "Instance" then
		if Objective:IsA("BasePart") then
			return (Objective.Position - RootPart.Position).Magnitude
		else
			return 0
		end
	elseif typeof(Objective) == "Vector3" then
		return (Objective - RootPart.Position).Magnitude
	elseif typeof(Objective) == "CFrame" then
		return (Objective.Position - RootPart.Position).Magnitude
	end
	return 0
end

local function InRaya(Path, condition, Base_position, mode, range)
    local results = {}
    local maxRange = math.huge
    local sortMode = "nearest"
    
    if type(mode) == "number" then
        maxRange = mode
        sortMode = "nearest"
    elseif mode == "farthest" then
        sortMode = "farthest"
    elseif mode == "value" then
        sortMode = "value"
    elseif type(range) == "number" then
        maxRange = range
    end

    for _, Object in pairs(Path) do
        if condition(Object) then
            if sortMode == "value" then
                local diff = math.abs(Object.price - Base_position)
                table.insert(results, { obj = Object, dist = diff })
            else
                local Distance = (Object:GetPivot().Position - Base_position).Magnitude
                if Distance <= maxRange then
                    table.insert(results, { obj = Object, dist = Distance })
                end
            end
        end
    end

    if #results == 0 then return nil end

    table.sort(results, function(a, b) return a.dist < b.dist end)
    return results[1].obj
end

local function tp(target)
    if Humanoid.Health <= 0 then return end
    if not RootPart then return end
    if Humanoid.Sit then Humanoid.Sit = false end

    -- ปิด collision กันติดกำแพงระหว่างบิน
    for _, part in ipairs(Character:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end

    -- ล็อคแกน Y กันร่วง (แบบ eiei.txt)
    local head = Character:FindFirstChild("Head")
    if head and not head:FindFirstChild("FlyLock") then
        local bv = Instance.new("BodyVelocity")
        bv.Name = "FlyLock"
        bv.MaxForce = Vector3.new(0, math.huge, 0)
        bv.Velocity = Vector3.zero
        bv.Parent = head
    end

    local targetPos = typeof(target) == "CFrame" and target.Position or target
    local currentDist = (targetPos - RootPart.Position).Magnitude

    if currentDist >= 2000 then
        local distm, middle = math.huge, nil
        for _, pos in pairs(CanTeleport[WorldSea]) do
            local d = (pos - targetPos).Magnitude
            if d < currentDist and d < distm then
                distm, middle = d, pos
            end
        end

        if middle then
            if Near_Island(RootPart.Position).Name ~= Near_Island(middle).Name then
                _C("requestEntrance", middle)
                task.wait(1)
            end
        end
    end

    local a = CFrame.new(targetPos)
    local W = (RootPart.CFrame.Position - a.Position).Magnitude
    local h = RootPart.CFrame
    RootPart.CFrame = CFrame.new(h.X, a.Y, h.Z)

    if W <= 5 then
        RootPart.CFrame = a
        return
    end

    local divisor = W < 100 and 250 or 240
    local duration = W / divisor

    if _G.TweenInstance then pcall(function() _G.TweenInstance:Cancel() end) end
    _G.TweenInstance = TweenService:Create(RootPart, TweenInfo.new(duration, Enum.EasingStyle.Sine), {CFrame = a})
    _G.TweenInstance:Play()
    _G.TweenInstance.Completed:Wait()
end

local function tp_boat(target, boat, etc, speeds)
	if not RootPart then return end 

	local startPos  = boat.VehicleSeat:GetPivot().Position
	local targetPos = target.Position
	local speed     = speeds or 250
	local dir       = (targetPos - startPos).Unit
	local dist      = (targetPos - startPos).Magnitude
	local movedDist = 0
	local startTime = tick()

	while movedDist < dist and Humanoid.Health > 0 and RootPart and not _G['StopTween'] and Humanoid.Sit and not etc() do
		VirtualInputManager:SendKeyEvent(true, 'W', false, game)
		local elapsed     = tick() - startTime
		movedDist         = math.min(elapsed * speed, dist)
		local newPos = Vector3.new(startPos.X + dir.X * movedDist,40,startPos.Z + dir.Z * movedDist)

		boat.VehicleSeat:PivotTo(CFrame.new(newPos))
		task.wait()
	end
	VirtualInputManager:SendKeyEvent(false, 'W', false, game)
end

local function IsAlive(model)
    if not model then return end
    local Humanoids = model:FindFirstChild("Humanoid")
    if not Humanoids then return end
    local RealHealth = 0
    if Humanoids:IsA('Humanoid') then  RealHealth = Humanoids.Health else  RealHealth = Humanoids.Value end
    local RootParts = model:FindFirstChild('HumanoidRootPart')
    if not RootParts then return end
    return RealHealth > 0
end

local BringModule = (function()
    local Cache = {
        _Tags = {},
        _EnableBring = true,
        _IgnoreName = true,
        _Distance = 270,
        _ActiveThreads = {},
        _LastUpdate = 0,
        _UpdateInterval = 0.1,
    }

    local Attachment = Instance.new("Attachment")
	local BRINGTAGS = "HEEDUM"
    do
        local AlignPosition = Instance.new("AlignPosition")
        AlignPosition.Mode = Enum.PositionAlignmentMode.OneAttachment
        AlignPosition.Position = Vector3.new(0, 20, 0)
        AlignPosition.Responsiveness = 200
        AlignPosition.MaxForce = 9e9
        AlignPosition.Parent = Attachment
        AlignPosition.Attachment0 = Attachment
    end

    local function New(Enemy)
        if not Enemy or not Enemy:FindFirstChild("HumanoidRootPart") then return end

        local self = Cache
        local Root = Enemy.HumanoidRootPart
        local Name = Enemy.Name
        local Distance = self._Distance
        local CloneAttachment = Attachment:Clone()
        CloneAttachment.Parent = Root

        local AlignPosition = CloneAttachment:FindFirstChildOfClass("AlignPosition")
        local isActive = true

        local threadId = tostring(Enemy)
        self._ActiveThreads[threadId] = true

        task.spawn(function()
            local lastUpdate = 0
            local updateRate = 0.05

            while isActive and IsAlive(Enemy)
                and Enemy.Parent == Enemies
                and CollectionService:HasTag(Enemy, BRINGTAGS) do

                local now = tick()
                if now - lastUpdate >= updateRate then
                    local Target = self._Tags[(self._IgnoreName and "ALL") or Name]
                    if Target and (Target.Position - Root.Position).Magnitude <= Distance then
                        AlignPosition.Position = Target.Position
                        lastUpdate = now
                    else
                        break
                    end
                end

                task.wait(updateRate)
            end

            isActive = false
            self._ActiveThreads[threadId] = nil

            if Enemy and CollectionService:HasTag(Enemy, BRINGTAGS) then
                CollectionService:RemoveTag(Enemy, BRINGTAGS)
            end

            if CloneAttachment and CloneAttachment.Parent then
                CloneAttachment:Destroy()
            end
        end)
    end

    function Cache:Bring(Goto, IgnoreName, CustomCFrame, Distance)
        if not Goto or not IsAlive(Goto) then return end

        local now = tick()
        if now - self._LastUpdate < self._UpdateInterval then return end
        self._LastUpdate = now

        local Root = Goto:FindFirstChild("HumanoidRootPart") or Goto.PrimaryPart
        if not Root then return end

        pcall(sethiddenproperty, Client, "SimulationRadius", math.huge)
        pcall(setscriptable, Client, "SimulationRadius", true)

        self._IgnoreName = IgnoreName ~= false
        local Name = Goto.Name
        local Position = Character:GetPivot().Position
        local PositionTag = self._IgnoreName and "ALL" or Name
        local Target = CustomCFrame or Root.CFrame
        local MaxDistance = Distance or self._Distance

        if not self._Tags[PositionTag]
        or (Target.Position - self._Tags[PositionTag].Position).Magnitude > 25 then
            self._Tags[PositionTag] = Target
        end

        for _, Enemy in ipairs(GetAllMonster()) do
            if (self._IgnoreName or Enemy.Name == Name)
            and not CollectionService:HasTag(Enemy, BRINGTAGS)
            and Enemy:FindFirstChild("Humanoid")
            and Enemy:FindFirstChild("HumanoidRootPart") then
                local ERoot = Enemy.HumanoidRootPart
                if IsAlive(Enemy) and (Position - ERoot.Position).Magnitude < MaxDistance then
                    Enemy.Humanoid.WalkSpeed = 0
                    Enemy.Humanoid.JumpPower = 0
					if not Enemy:FindFirstChild('HumanoidRootPart'):FindFirstChild('BodyVelocity') then 
						local Noclip         = Instance.new("BodyVelocity")
						Noclip.Parent        = Enemy:FindFirstChild('HumanoidRootPart')
						Noclip.MaxForce      = Vector3.new(100000, 100000, 100000)
						Noclip.Velocity      = Vector3.new(0, 0, 0)
					end
                    CollectionService:AddTag(Enemy, BRINGTAGS)
                end
            end
        end
    end

    function Cache:Stop()
        self._EnableBring = false
        self._ActiveThreads = {}
        self._Tags = {}
        for _, Enemy in ipairs(GetAllMonster()) do
            if CollectionService:HasTag(Enemy, BRINGTAGS) then
				if Enemy:FindFirstChild('HumanoidRootPart'):FindFirstChild('BodyVelocity') then 
					Enemy:FindFirstChild('HumanoidRootPart'):FindFirstChild('BodyVelocity'):Destroy()
				end
                CollectionService:RemoveTag(Enemy, BRINGTAGS)
            end
        end
    end

    CollectionService:GetInstanceAddedSignal(BRINGTAGS):Connect(New)

    return Cache
end)()

local function BringMob(m, ignoreName, customCFrame, distance)
    BringModule:Bring(m, ignoreName, customCFrame, distance)
end

		repeat
			local CFrame_Custom = (type(d) == "function" and d()) or d or CFrame.new(0,40,0)
			if bringMob then BringMob(v) end
			tp(v.HumanoidRootPart.CFrame * CFrame_Custom)
			Equip(GetToolFromTip("Melee"))

local function GetQuest()
	local QuestReal, dist, sub, Position_Mon, Keys = nil, -math.huge, nil, nil, nil

	for i, v in pairs(QuestData) do
		for s, x in pairs(v) do
			if x.LevelReq then
				if Client.Data.Level.Value >= x.LevelReq then
					if x.LevelReq > dist then
						dist      = x.LevelReq
						QuestReal = x
						sub       = i
						Keys      = s
					end
				end
			end
		end
	end

	local Name_Mon  = next(QuestReal.Task)
	local QuestNpc  = GuideModule.GetNearestNPC(GuideModule, Client.Character.HumanoidRootPart.Position, Client.Data.Level.Value)[1]

	return {
		[1] = QuestReal,
		[2] = Keys,
		[3] = dist,
		[4] = QuestNpc,
		[5] = sub,
		[6] = Position_Mon,
		[7] = Name_Mon
	}
end

local function CheckCooldown(Name, Skill)
	if Name then
		local Skills    = PlayerGui.Main.Skills
		local Weapon    = Skills:FindFirstChild(tostring(Name))
		local SkillFound = Weapon and Weapon:FindFirstChild(Skill)
		return Skill and SkillFound.Cooldown.AbsoluteSize.X ~= 0
	end
end

end

local function GetTableIndex(index)
	local Turtle = Maps and Maps:FindFirstChild("Turtle")
	if not Turtle or not index then return; end

	local TableTrade = {}
	local i = 1

	for _,v in ipairs(Turtle:GetChildren()) do
		if v.Name == "TradeTable" then
			TableTrade["Table"..i] = v
			i += 1
		end
	end

	return TableTrade[index]
end

local MM_request = http_request or request or (syn and syn.request)
local MM_SERVER_URL = "http://168.222.28.44:3000"  -- << IP VPS ของคุณ
local MM_MyId = Client.UserId

local function MM_HasOrangeBelt()
	local ok, inv = pcall(function()
		return ReplicatedStorage.Remotes.CommF_:InvokeServer("getInventory")
	end)
	if not ok or not inv then return false end
	for _, v in pairs(inv) do
		if v.Name == "Dojo Belt (Orange)" then return true end
	end
	return false
end

local function MM_Announce(hasOrange)
	pcall(function()
		MM_request({
			Url = MM_SERVER_URL .. "/announce",
			Method = "POST",
			Headers = { ["Content-Type"] = "application/json" },
			Body = HttpService:JSONEncode({
				jobId = game.JobId,
				userId = MM_MyId,
				name = Client.Name,
				hasOrange = hasOrange,
			})
		})
	end)
end

local function MM_FetchPartner()
	local ok, res = pcall(function()
		return MM_request({
			Url = MM_SERVER_URL .. "/partner?jobId=" .. game.JobId .. "&me=" .. MM_MyId,
			Method = "GET",
		})
	end)
	if ok and res and res.Body and res.Body ~= "" then
		local good, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
		if good and data and data.userId then return data end
	end
	return nil
end

local function Section(Tab, Text)
	return Tab:Section({
		Title    = Text,
		Box      = false,
		TextSize = 17,
		Opened   = true,
	})
end

local function Tg(def, Tab, Description, Title, Keys, Callback)
	local taskThread
	local toggle = Tab:Toggle({
		Title    = Title,
		Desc     = Description,
		Value    = c()[Keys] or def,
		Callback = function(state)
			Config_Manager.S()[Keys] = state
			c()[Keys] = state
			if Callback and typeof(Callback) == "function" then
				Callback(state)
			end
			if not state then
				if taskThread then
					task.cancel(taskThread)
					taskThread = nil
				end
			end
			if state then
				if func[Keys] then
					taskThread = task.spawn(function() func[Keys]() end)
				end
			end
			Config_Manager.SG()
		end
	})
	if toggle then toggle.Callback(c()[Keys]) end
	return toggle
end

local function Slider(Tab, Title, Steps, Mins, Maxs, Def, Keys)
	local slider = Tab:Slider({
		Flag     = Title,
		Title    = Title,
		Step     = Steps,
		Value    = {
			Min     = Mins,
			Max     = Maxs,
			Default = c()[Keys] or Def,
		},
		Callback = function(value)
			Config_Manager.S()[Keys] = value
			c()[Keys] = value
			Config_Manager.SG()
		end
	})
	if slider then slider.Callback(c()[Keys] or Def) end
	return slider
end

local function Input(Tab, Title, Keys)
	return Tab:Input({
		Title       = Title,
		Type        = "Input",
		Placeholder = "Enter here...",
		Callback    = function(value)
			Config_Manager.S()[Keys] = value
			c()[Keys] = value
			Config_Manager.SG()
		end
	})
end

local function Dropdown(Tab, Title, Keys, Values, Multi)
	local dropdown = Tab:Dropdown({
		Title      = Title,
		Values     = Values or {"nigger", "Back"},
		Multi      = Multi or false,
		AllowNone  = false,
		Value      = c()[Keys] or Values[1],
		Callback   = function(option)
			Config_Manager.S()[Keys] = option
			c()[Keys] = option
			Config_Manager.SG()
		end
	})
	if dropdown and dropdown.Callback then
		dropdown.Callback(c()[Keys] or (Multi and {} or Values[1]))
	end
	return dropdown
end

local function Button(Tab, Title, Desc, Callback)
	return Tab:Button({
		Title    = Title,
		Desc     = Desc,
		Locked   = false,
		Callback = function()
			if Callback then Callback() end
		end
	})
end
